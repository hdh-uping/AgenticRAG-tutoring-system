int DeQueue(LinkQueue *q, Elemtype &e)
{
    //若队列不为空将出队，即为删除队头结点
    QueueNode *p;
    if(QueueEmpty(q)) return ERROR;    //队空
    p=q->front->next;
    e=p->data;
    q->front->next=p->next;
    free(p);
    if(q->front->next==NULL) q->rear=q->front;
    return OK;
}
