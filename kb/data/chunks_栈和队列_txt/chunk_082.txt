```c
LinkQueue* InitQueue()
//创建一个带头结点的空队
{
    LinkQueue *q;
    QueueNode *p;
    q=(LinkQueue*)malloc(sizeof(LinkQueue)); //申请头尾指针结点
    p=(QueueNode*)malloc(sizeof(QueueNode)); //申请链队列头结点
    p->next=NULL;
    q->front=q->rear=p;
    return q;
}
```
