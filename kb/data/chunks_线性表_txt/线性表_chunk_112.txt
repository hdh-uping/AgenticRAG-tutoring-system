/*在顺序线性表L中查找第1个值与e相等的元素的位序，若找到，则返回其在L中的位序，否则返回-1*/
int LocateElem_Sq(SqList L, ElemType e)
{
    int i=0;    /*i 的初值为第一个元素的位序*/
    while(i<L.length&&L.elem[i]!=e) i++;
    if(L.elem[i]==e) return i;
}
